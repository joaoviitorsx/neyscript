// Estatisticas e jogadas do camisa 10
// ============================================================

/*  tipos:
    camisa10    -> inteiro
    ousadia     -> ponto flutuante
    canto     -> texto (string)
    semGol      -> sem retorno
*/

// partida (funcao) que soma gols de duas temporadas
partida camisa10 somarGols(camisa10 temporadaA, camisa10 temporadaB) {
    camisa10 total = temporadaA + temporadaB;
    gol total;
}

// partida que calcula a media de gols por jogo
partida ousadia mediaGols(camisa10 gols, camisa10 jogos) {
    ousadia media = 0.0;
    seJogar (jogos > 0) {
        media = gols / jogos;
    } banco {
        media = 0.0;
    }
    gol media;
}

// partida sem retorno - apenas imprime o nome do craque
partida semGol anunciar(canto nome) {
    narra("Em campo: ");
    narra(nome);
    gol ;
}

// ponto de entrada do programa
neymar() {

    canto  jogador    = "Neymar Jr";
    camisa10 golsPSG    = 118;
    camisa10 golsSantos = 136;
    camisa10 jogos      = 250;
    camisa10 totalGols  = 0;
    ousadia  media      = 0.0;

    // chamadas de partida com 'assistencia' (aridade e tipos conferidos)
    totalGols = assistencia somarGols(golsPSG, golsSantos);
    media     = assistencia mediaGols(totalGols, jogos);

    // chamada de partida sem retorno como statement
    assistencia anunciar(jogador);

    // saida com narra
    narra("Gols totais: ");
    narra(totalGols);
    narra("Media por jogo: ");
    narra(media);

    // condicional
    seJogar (totalGols >= 250) {
        narra("LENDA DO FUTEBOL!");
    } banco {
        narra("ainda em construcao da lenda");
    }

    // laco de repeticao
    camisa10 contador = 0;
    replay (contador < 3) {
        narra("OLE OLE OLE NEYMAR");
        contador = contador + 1;
    }

    // concatenacao de canto (string) com '+'
    canto grito = "OLE" + "OLE";
    narra(grito);

    // coercao implicita camisa10 -> ousadia
    ousadia ajuste = totalGols * 1.5;
    narra(ajuste);

    // rodada (for): conta as 5 melhores temporadas
    rodada (camisa10 temp = 1; temp <= 5; temp = temp + 1) {
        narra("temporada ");
        narra(temp);
        seJogar (temp == 3) {
            driblar;    // pula temporada 3 (lesao)
        }
    }

    // apito (break): encerra replay quando meta atingida
    camisa10 meta = 0;
    replay (meta < 10) {
        meta = meta + 1;
        seJogar (meta == 7) {
            apito;      // chegou a 7, encerra
        }
    }

    // lesionado (null): variavel canto sem valor inicial
    canto apelido = lesionado;
    apelido = "Ney";
    narra(apelido);

    // literais booleanos campeao/vice
    camisa10 titulo = campeao;
    camisa10 derrota = vice;
    seJogar (titulo) {
        narra("CAMPEAO DO MUNDO!");
    } banco {
        narra("vice...");
    }

    gol totalGols;
}
